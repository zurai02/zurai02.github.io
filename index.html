LuauSignal.lua
-- lightweight event signal
local Signal = {}
Signal.__index = Signal

function Signal.new()
  return setmetatable({
    listeners = {},
  }, Signal)
end

function Signal:Connect(callback)
  local connection = {
    Connected = true,
    Callback = callback,
  }
  table.insert(self.listeners, connection)
  return connection
end
